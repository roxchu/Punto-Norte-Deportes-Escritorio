//FUNCION PRINCIPAL
int main() {
    //DECLARACION DE VARIABLES
    int op;
    Libros *libro=NULL;
    Usuarios *usuario=NULL;
    Prestamos *prestamo=NULL;
    //CUERPO DEL PROGRAMA
    do{
        op=menu();//LLAMADO A FUNCION MENU
        switch (op){//DE ACUERDO A OP SE LLAMAN A FUNCIONES
            case 1:
                regLibro(libro);
                break;
            case 2:
                regUsuario(usuario);
                break;
            case 3:
                consLibro(libro);
                break;
            case 4:
                if (libro!=NULL && usuario!=NULL){
                    prestar(prestamo,libro,usuario);
                }else{
                    cout<<"Es necesario cargar usuarios y/o libros";
                }
                break;
            case 5:
                if (prestamo!=NULL){
                    devolver(prestamo);
                }else{
                    cout<<"NO HAY PRESTAMOS REALIZADOS";
                }
                break;
            case 6:
                if (usuario!=NULL){
                    histUsuario(prestamo,usuario);
                }else{
                    cout<<"NO HAY USUARIOS CARGADOS";
                }
                break;
            case 7:
                break;
            default:
                cout<<"Ingrese opcion valida";
        }
    }while (op!=7);
}
//REGISTRO DE LIBROS
void regLibro(Libros *&lib){
    Libros *nuevo;
    Libros *aux;
    string rta;
    cin.ignore();//LIMPIEZA DE BUFFER DE ENTRADA OBLIGATORIA ANTES DE CADA GETLINE QUE VIENE DE UN CIN(EN ESTE CASO EL DE LA FUNCION PRINCIPAL) 
    do{//COMIENZA CICLO DE REPETICION
        nuevo = new Libros;
        cout<<"Ingrese Titulo del Libro: ";
        //COMO DENTRO DEL CICLO LA ULTIMA ENTRADA ES UN GETLINE EL CIN.IGNORE VA FUERA DEL CICLO
        getline(cin, nuevo->titulo);
        cout<<"Ingrese Autor del Libro: ";
        getline(cin, nuevo->autor);
        cout<<"Ingrese Año de publicacion del Libro: ";
        cin>>nuevo->anio_pub;
        cout<<"Ingrese Cantidad de ejemplares del Libro: ";
        cin>>nuevo->ejemplares;
        if(lib == NULL){//SI ES EL PRIMER LIBRO QUE SE REGISTRA SE LE DA EL ID 1
            nuevo->id = 1;
            lib = nuevo;
        }else{//SI HAY OTRO LIBRO REGISTRADO
            aux = lib;//USANDO UN PUNTERO AUXILIAR
            while(aux->sgte != NULL){//VAMOS AL ULTIMO
                aux = aux->sgte;
            }
            nuevo->id = aux->id + 1;//AL LIBRO A INGRESAR LE DAMOS EL ULTIMO ID + 1
            aux->sgte=nuevo;//LO AGREGAMOS A LA LISTA
        }
        nuevo->sgte = NULL;//PONEMOS EL ULTIMO SGTE EN NULL
        cout<<"S para seguir cargando: ";
        cin.ignore();
        getline(cin,rta);
    }while(rta=="S" || rta=="s");
}
//REGISTRO DE USUARIOS

            nuevo->id = 1;

        }else{

        }

    }while(rta=="S" || rta=="s");
}
//MOSTRAR LIBROS CARGADOS
void consLibro(Libros *libro){
    if (libro==NULL){//SI AUN NO SE CARGARON LIBROS SE AVISA
        cout<<"NO HAY LIBROS CARGADOS\n";
    }else{//SI SE CARGARON EMPEZAMOS A MOSTRAR
        cout<<setw(50)<<"LIBROS"<<endl;//TITULO DE LA TABLA
        cout<<string(100,'_')<<endl;//LINEA DE ARRIBA DE LA TABLA
        //COLUMNAS setw(x) alinea a la derecha reservando x lugares para la palabra EJEMPLO setw(10)<<"ID" deja 8 lugares libres y ID suma 10
        cout<<'|'<<setw(10)<<"ID"<<setw(20)<<"Titulo"<<setw(20)<<"Autor"<<setw(20)<<"Publicado"<<setw(20)<<"Ejemplares"<<setw(10)<<'|'<<endl;
        while (libro!=NULL){//mientras haya libros los muestra
            cout<<"|"<<string(99,'-')<<"|"<<endl;//string(99,'-') escribe 99 veces -
            cout<<'|'<<setw(10)<<libro->id<<setw(20)<<libro->titulo<<setw(20)<<libro->autor<<setw(20)<<libro->anio_pub<<setw(20)<<libro->ejemplares<<setw(10)<<'|'<<endl;
            libro=libro->sgte;//pasa al siguiente libro
        }
        cout<<"|"<<string(99,'_')<<"|"<<endl;
    }
}

void prestar(Prestamos *&prestamo, Libros *libro,Usuarios *usuario){
    Prestamos *nuevo;
    Prestamos *aux;
    string rta;
    do{
    nuevo = new Prestamos;
    consUsuario(usuario);
    cout<<"Ingrese Id del Usuario: ";
    cin>>nuevo->usuario;
    consLibro(libro);
    cout<<"Ingrese Id del Libro: ";
    cin>>nuevo->libro;
    cout<<"Ingrese fecha inicio del prestamo (DD/MM/AA): ";
    cin.ignore();
    getline(cin,nuevo->fprestamo);
    cout<<"Ingrese fecha inicio del prestamo (DD/MM/AA): ";
    getline(cin,nuevo->fdevolucion);

    cout<<"S para seguir cargando: ";

    }while(rta=="S" || rta=="s");

void devolver(Prestamos *&prestamo){
    int idprest;
    Prestamos *aux=prestamo;
    consPrest(prestamo);
    cout<<"Cual prestamo finalizo? 0 si ninguno";
    cin>>idprest;
    while (aux->id!=idprest){
        aux=aux->sgte;
    }
    aux->volvio=true;
}
